# PyInstaller version info — попадает в exe как VS_VERSION_INFO ресурс.
# Windows использует эти поля в Properties → Details и в UAC-prompt.
# ProductName/CompanyName также помогают антивирусам опознавать подписанный exe.
VSVersionInfo(
    ffi=FixedFileInfo(
        # ProductVersion и FileVersion — четверные (major.minor.patch.build).
        # Берём 2026.10.1.0.
        filevers=(2026, 10, 1, 0),
        prodvers=(2026, 10, 1, 0),
        mask=0x3F,
        flags=0x0,
        OS=0x40004,            # VOS_NT_WINDOWS32
        fileType=0x1,          # VFT_APP
        subtype=0x0,
        date=(0, 0),
    ),
    kids=[
        StringFileInfo([
            StringTable(
                "040904B0",    # LangID 1033 (en-US), CharsetID 1200 (UTF-16)
                [
                    StringStruct("CompanyName",      "Aleksey Blagonravov"),
                    StringStruct("FileDescription",  "Masking — локальный сервис маскирования ПДн"),
                    StringStruct("FileVersion",      "2026.10.1"),
                    StringStruct("InternalName",     "masking-service"),
                    StringStruct("LegalCopyright",   "© 2026 Aleksey Blagonravov. MIT License."),
                    StringStruct("OriginalFilename", "masking-service.exe"),
                    StringStruct("ProductName",      "Masking"),
                    StringStruct("ProductVersion",   "2026.10.1"),
                ],
            ),
        ]),
        VarFileInfo([VarStruct("Translation", [1033, 1200])]),
    ],
)
